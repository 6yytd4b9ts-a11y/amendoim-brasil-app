// Exportação de um período em casca equivalente (mil t): amendoim (em casca + grão ÷ fator) e óleo (÷ fator).
export function exportEmCasca(x, X) {
  if (!x) return null;
  return { amend: (x.casca || 0) + (x.grao || 0) / (X.fatorGrao || 0.7), oleo: (x.oleo || 0) / (X.fatorOleo || 0.38) };
}
// Balanço por exclusão das safras que têm exportação (as duas últimas), em mil t em casca. Safra parcial: exportação
// levada para 12 meses no ritmo atual. A sobra de uma safra entra como estoque inicial da seguinte.
export function balancoSafras(c, X) {
  X = X || c.exportCasca || PADRAO.exportCasca;
  const b = { ...PADRAO.balanco, ...(c.balanco || {}) }, lista = c.estimativas || [];
  const consumo = (b.consumoKgHab * b.populacaoMi) / ((b.rendimento || 70) / 100);
  const cols = [];
  let ant = null;
  lista.forEach((e, i) => {
    const x = X.safras?.[e.safra];
    if (!x) { ant = null; return; }
    const ex = exportEmCasca(x, X), meses = x.meses || 12, k = 12 / meses, prox = lista[i + 1];
    const col = { safra: e.safra, periodo: x.periodo, meses, parcial: meses < 12, inicial: ant ? ant.sobra : null, producao: producao(e).tMeio / 1000,
      amend: ex.amend * k, oleo: ex.oleo * k, consumo, semente: prox ? (prox.areaMin * b.sementeKgHa) / 1000 : 0 };
    col.sobra = (col.inicial || 0) + col.producao - col.amend - col.oleo - col.consumo - col.semente;
    cols.push(col); ant = col;
  });
  return { cols: cols.slice(-2), X, b };
}

function resumoBrasil(b) {
  const atual = b.ano, n = b.mes, ate = MESES[n - 1].toLowerCase();
  const s = (p, a) => (b.series[p]?.[a]?.t || []).slice(0, n).reduce((x, v) => x + (v || 0), 0);
  const casca = (a) => s('grao', a) / RENDIMENTO + s('oleo', a) / FATOR_OLEO;
  const lin = (rot, sub, u, a) => `<tr><th>${rot}${sub ? `<br><small>${sub}</small>` : ''}</th><td class="num">${nf(a)} t</td><td class="num">${nf(u)} t</td><td class="dx-var">${vc(varia(u, a))}</td></tr>`;
  return `<section class="cartao dx-cartao" style="gap:6px">
      <div class="cartao-cab"><span class="rotulo">Exportações jan–${ate}</span><span class="mini">${atual - 1} × ${atual}</span></div>
      <table class="tabela dx-tabela"><thead><tr><th></th><th>${atual - 1}</th><th>${atual}</th><th>Var.</th></tr></thead><tbody>
        ${lin('<b>Total em casca</b>', 'grão ÷ 0,70 + óleo ÷ 0,38', casca(atual), casca(atual - 1))}
        ${lin('Grão', 'NCM 1202', s('grao', atual), s('grao', atual - 1))}
        ${lin('Óleo', 'NCM 1508', s('oleo', atual), s('oleo', atual - 1))}
      </tbody></table>
      <span class="mini">Total em casca: quanto amendoim da roça a exportação consome. Em sacas de 25 kg: ${nf((casca(atual - 1) * 40) / 1e6, 1)} mi em ${atual - 1} e ${nf((casca(atual) * 40) / 1e6, 1)} mi em ${atual}.</span>
    </section>`;
}

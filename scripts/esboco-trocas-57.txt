// Miniatura da notícia: a foto, se tiver; senão a bandeira do país, um ícone de clima ou o símbolo do app.
function miniNoticia(n, h) {
  const caixa = (dentro, fundo = 'var(--fundo)') => `<span aria-hidden="true" style="flex:none;width:42px;height:42px;border-radius:10px;overflow:hidden;display:flex;align-items:center;justify-content:center;background:${fundo}">${dentro}</span>`;
  const img = n.imagem || n.capa;
  if (/^https:\/\//.test(img || '')) return caixa(`<img src="${h.esc(img)}" alt="" loading="lazy" style="width:100%;height:100%;object-fit:cover">`);
  const t = `${n.titulo || ''} ${n.fonte || ''}`;
  if (/chuva|clima|calor|frio|geada|seca|Climatempo|Cepagri|INMET|previs/i.test(t)) return caixa(h.ic(h.I.chuva, 'style="width:24px;height:24px;stroke:#2F6FA3"'), '#E6F0F8');
  const pais = /EUA|Estados Unidos|USDA|american/i.test(t) ? 'US' : /Argentin|Córdoba|Cordoba|Maní/i.test(t) ? 'AR' : /Índia|India/i.test(t) ? 'IN' : /China|chin[eê]s/i.test(t) ? 'CN' : /Conab|IEA|Brasil|paulista|SP\b|Comex/i.test(t) ? 'BR' : '';
  if (pais) return caixa(bandeira(pais, 42));
  return caixa('<img src="/icons/icon-maskable-512.png" alt="" style="width:100%;height:100%;object-fit:cover">');
}

    (t.fatores || []).forEach((f) => { if (FAT[f.nome]) Object.assign(f, FAT[f.nome]); });

  (D.mercado?.hoje || []).forEach((x) => { if (x.id === 'exportacao') Object.assign(x, { valor: 'Mais fraca', efeito: 'baixa', nota: 'grão +16% · óleo −22%', detalhe: expDet }); });

}

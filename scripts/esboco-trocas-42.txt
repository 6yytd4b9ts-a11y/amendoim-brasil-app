function ultimoAviso() {
  const a = est.dados?.ultimoAviso;
  if (!a) return '';
  const m = {}; String(a.detalhe || '').split(' ').forEach((p) => { const i = p.indexOf('='); if (i > 0) m[p.slice(0, i)] = p.slice(i + 1); });
  const ent = Number(m.entregues || 0);
  const txt = m.erro ? 'não saiu (falha no envio)' : ent > 0 ? `chegou em ${ent} celular${ent > 1 ? 'es' : ''}` : 'não chegou a nenhum celular: ligue os avisos de novo neste aparelho';
  return `<span class="mini" style="${ent > 0 ? '' : 'color:#B3261E;font-weight:700'}">Último aviso de pedido (${quando(a.criado_em, '')}): ${txt}.</span>`;
}

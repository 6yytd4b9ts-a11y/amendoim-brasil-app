function contextoConab(c) {

function ligarPuxar() {
  let y0 = null, dy = 0, pill = null;
  const topo = () => (document.scrollingElement || document.documentElement).scrollTop <= 0;
  const faz = () => { if (!pill) { pill = document.createElement('div'); pill.className = 'puxar-pill'; document.body.appendChild(pill); } return pill; };
  document.addEventListener('touchstart', (e) => { y0 = topo() && e.touches.length === 1 ? e.touches[0].clientY : null; dy = 0; }, { passive: true });
  document.addEventListener('touchmove', (e) => {
    if (y0 == null) return;
    dy = Math.max(0, e.touches[0].clientY - y0);
    if (!dy) return;
    const p = faz(); p.textContent = dy > 70 ? '↻ Solte para atualizar' : '↓ Puxe para atualizar'; p.style.transform = `translate(-50%,${Math.min(dy * 0.5, 54) - 8}px)`;
  }, { passive: true });
  document.addEventListener('touchend', () => {
    if (y0 == null) return;
    const ok = dy > 70; y0 = null;
    if (!pill) return;
    if (!ok) { pill.style.transform = 'translate(-50%,-70px)'; return; }
    pill.textContent = '↻ Atualizando…'; pill.style.transform = 'translate(-50%,12px)';
    setTimeout(() => location.reload(), 250);
  });
}

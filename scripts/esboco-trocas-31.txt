// Página de notícias (dentro de Mercado): o fato do dia com o texto inteiro e todas as notícias publicadas, cada uma abrindo o texto e a fonte.
export function telaNoticias(D, h) {
  const todas = (conteudo().noticias || []).filter((n) => n.titulo).sort((a, b) => (b.dia > a.dia ? 1 : b.dia < a.dia ? -1 : 0));
  const hoje = new Date().toISOString().slice(0, 10);
  const dia = (d) => (!d ? '' : d === hoje ? 'Hoje' : `${d.slice(8, 10)}/${d.slice(5, 7)}/${d.slice(0, 4)}`);
  const m = D.mercado;
  const grupos = [];
  todas.forEach((n) => { const g = grupos[grupos.length - 1]; if (g && g.dia === n.dia) g.itens.push(n); else grupos.push({ dia: n.dia, itens: [n] }); });
  const fonteLink = (fonte, link) => (/^https?:/.test(link || '') ? `<a href="${h.esc(link)}" target="_blank" rel="noopener">${h.esc(fonte || link)}</a>` : h.esc(fonte || ''));
  return `${m?.fato ? `<section class="cartao es-fato"><div class="fato"><span class="fato-tag">Fato do dia</span><b>${h.esc(m.fato.titulo)}</b><span>${h.esc(m.fato.texto || '')}</span>${m.fato.fonte || m.fato.link ? `<small class="mini">Fonte: ${fonteLink(m.fato.fonte, m.fato.link)}</small>` : ''}<button class="link-enviar" data-compartilhar="fato">${h.ic(h.I.enviar)}Enviar no WhatsApp</button></div></section>` : ''}
  ${grupos.length ? grupos.map((g) => `<span class="rotulo" style="padding:4px 4px 0">${dia(g.dia)}</span>
    ${g.itens.map((n) => { const href = /^https?:\/\//.test(n.link || '') ? n.link : ''; return `<details class="cartao nt-item" style="gap:6px">
      <summary style="cursor:pointer;list-style:none;display:flex;gap:10px;align-items:flex-start"><span class="es-not-ponto" style="margin-top:7px"></span><span class="cresce"><b style="font-size:15px;line-height:1.3;display:block">${h.esc(n.titulo)}</b>${n.fonte ? `<small class="mini">${h.esc(n.fonte)}</small>` : ''}</span><span aria-hidden="true" style="font-size:18px;color:var(--texto-3)">⌄</span></summary>
      <div class="es-txt" style="padding-left:18px">${n.texto ? h.esc(n.texto) : n.resumo ? h.esc(n.resumo) : '<span class="mini">Sem texto ainda.</span>'}</div>
      ${n.fonte || href ? `<span class="mini" style="padding-left:18px">Fonte: ${fonteLink(n.fonte, href)}</span>` : ''}
      ${href ? `<a class="btn btn-pequeno es-btn-claro" href="${h.esc(href)}" target="_blank" rel="noopener" style="margin-left:18px;align-self:flex-start">Ler na fonte</a>` : ''}
    </details>`; }).join('')}`).join('') : '<div class="vazio">Nenhuma notícia publicada ainda.</div>'}
  <p class="mini dx-nota">Notícias selecionadas e resumidas pela Amendoim Brasil a partir das fontes indicadas.</p>`;
}

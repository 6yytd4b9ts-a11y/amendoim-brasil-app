// iPhone: na primeira vez que se vira o celular o Safari às vezes mantém a escala da tela em pé e a tabela fica grande demais.
// Ao virar, a escala é fixada em 1 por um instante e depois solta de novo, então o zoom continua livre para o usuário.
(() => {
  const m = document.querySelector('meta[name=viewport]');
  if (!m) return;
  const livre = m.content; let t;
  const acertar = () => { m.content = 'width=device-width, initial-scale=1, minimum-scale=1, maximum-scale=1, viewport-fit=cover'; clearTimeout(t); t = setTimeout(() => { m.content = livre; }, 400); };
  addEventListener('orientationchange', acertar);
  try { screen.orientation?.addEventListener('change', acertar); } catch (e) { /* sem suporte */ }
})();
